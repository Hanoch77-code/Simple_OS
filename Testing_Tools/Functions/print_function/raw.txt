        mov dx,bx
        mov bx,hex
        add bx,5
        mov ch,4
    .repeat:
        mov ax,dx
        and ax,0x000F
        inc cl
        cmp ax,9
        jle .skip
        add ax,0x7
    .skip:
        add ax,0x30
        mov [bx],al
        dec bx

        shr dx,4

        dec ch
        jnz .repeat
